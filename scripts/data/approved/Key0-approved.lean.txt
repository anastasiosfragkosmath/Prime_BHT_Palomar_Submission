/-- The admissible smooth cutoff of the paper (lines 372-376): an even `C^∞` function whose topological
support lies in `(-1, 1)` and which equals `1` on `[-1/2, 1/2]`. (`∞` is `C^∞` smoothness, not analytic.) -/
def IsCutoff (χ : ℝ → ℝ) : Prop :=
  ContDiff ℝ ∞ χ ∧ (∀ t, χ (-t) = χ t) ∧ tsupport χ ⊆ Set.Ioo (-1) 1 ∧
    ∀ t ∈ Set.Icc (-1 / 2 : ℝ) (1 / 2), χ t = 1

/-- `ψ(t) = (χ(t) - χ(2t)) / t` with `ψ(0) = 0` (line 378; Lean's `x / 0 = 0` gives `ψ(0) = 0`). -/
noncomputable def psi (χ : ℝ → ℝ) (t : ℝ) : ℝ := (χ t - χ (2 * t)) / t

/-- `ψ_k(t) = 2^{-k} ψ(t 2^{-k})` (line 383). -/
noncomputable def psiK (χ : ℝ → ℝ) (k : ℕ) (t : ℝ) : ℝ := psi χ (t / 2 ^ k) / 2 ^ k

/-- The unsmoothed truncated long-scale operator (lines 625-627, with `a_Q = ⌈Q^τ⌉` of line 678)
`U^J_{Q;τ}(f, g)(x) = ∑_{Q^τ ≤ k ≤ J} ∑_n ψ_k(n) w_Q(n) f(x - n) g(x + n)` for finitely supported
`f, g`, where `w_Q(n) = Λ_Q(|n|)` is `PrimeCarleson.blockWeight (Finset.Ioc (Q / 2) Q) n` (the paper's
`Λ_Q` of line 351, even in `n`). For a cutoff `χ` the `n`-family has finite support, so `∑'` is the
genuine finite sum. -/
def uOp (χ : ℝ → ℝ) (Q : ℕ) (τ : ℝ) (J : ℕ) (f g : ℤ →₀ ℂ) (x : ℤ) : ℂ :=
  ∑ k ∈ Finset.Icc ⌈(Q : ℝ) ^ τ⌉₊ J,
    ∑' n : ℤ, ((psiK χ k (n : ℝ) : ℝ) : ℂ) *
      PrimeCarleson.blockWeight (Finset.Ioc (Q / 2) Q) n * f (x - n) * g (x + n)
